cmake_minimum_required(VERSION 3.22)
project(QwenNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/stable-diffusion.cpp)
add_library(qwen_bridge STATIC Native/QwenBridge.cpp)
target_include_directories(qwen_bridge PUBLIC Native/include)
target_link_libraries(qwen_bridge PUBLIC stable-diffusion)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_testing()
    add_executable(bridge_smoke Native/BridgeSmoke.cpp)
    target_link_libraries(bridge_smoke PRIVATE qwen_bridge)
    add_test(NAME bridge_smoke COMMAND bridge_smoke)
endif()
